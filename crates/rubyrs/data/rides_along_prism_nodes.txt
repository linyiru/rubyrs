# Prism node classes that ast.rs does NOT visit via an `as_*_node`
# accessor, but whose contents are consumed inline by a parent node
# that IS supported. For gapscan classification, these count as
# "translatable" — they don't represent gaps.
#
# Each entry must justify why it rides along.
#
# Format: one CamelCase name per line, sorted, `#` comments OK.

# def.parameters() returns ParametersNode; ast.rs walks its
# requireds/optionals directly without as_parameters_node.
ParametersNode

# call.arguments() / yield.arguments() / return.arguments() etc.
# all return ArgumentsNode; ast.rs iterates a.arguments() inline.
ArgumentsNode

# begin.rescue_clause() returns RescueNode; ast.rs walks the
# rescue chain via subsequent() without as_rescue_node.
RescueNode

# begin.ensure_clause() returns EnsureNode; ast.rs reads its
# statements directly.
EnsureNode
